{{- /* Resize images stored with a post and wrap titled images in a figure. */ -}} {{- $src := .Destination -}} {{- $w := "" }}{{ $h := "" -}} {{- with .Page.Resources.Get .Destination -}} {{- $img := . -}} {{- if gt .Width 1400 }}{{ $img = .Resize "1400x" }}{{ end -}} {{- $src = $img.RelPermalink }}{{ $w = $img.Width }}{{ $h = $img.Height -}} {{- end -}} {{- $tag := printf "" $src .Text (cond (ne $w "") (printf " width=\"%d\" height=\"%d\"" $w $h) "") -}} {{- if .Title -}} {{- else -}} {{ $tag | safeHTML }} {{- end -}}
